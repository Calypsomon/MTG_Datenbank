TABLE card(
    -- ===== BASIC =====
    id TEXT PRIMARY KEY,
    private_id INTEGER,
    name TEXT,
    lang TEXT,
    count INTEGER,

    -- ===== SET INFO =====
    set TEXT,
    set_name TEXT,
    set_id TEXT,
    collector_number TEXT,
    released_at DATE,

    -- ===== GAME DATA =====
    type_line TEXT,
    oracle_text TEXT,
    mana_cost TEXT,
    cmc DOUBLE PRECISION,

    colors TEXT[],
    color_identity TEXT[],
    keywords TEXT[],

    rarity TEXT,
    layout TEXT,

    -- ===== FLAGS =====
    reserved BOOLEAN,
    reprint BOOLEAN,
    full_art BOOLEAN,
    booster BOOLEAN,

    -- ===== META =====
    border_color TEXT,
    edhrec_rank INTEGER,

    -- ===== ART =====
    artist TEXT,
    card_back_id TEXT,

    -- ===== SIMPLIFIED OPTIONAL =====
    price_eur DOUBLE PRECISION,
    image_png TEXT,
    image_small TEXT,
    image_normal TEXT,
    image_png_data BYTEA,
    image_small_data BYTEA,
    image_normal_data BYTEA,
    purchase_cardmarket TEXT
);
